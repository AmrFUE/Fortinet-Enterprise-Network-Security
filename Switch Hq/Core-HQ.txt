enable
configure terminal

hostname Core-HQ

! Create VLANs
vlan 10
 name Users
vlan 20
 name Servers
vlan 30
 name Management
exit

! Configure all 4 ports as 802.1Q Trunks
! Gi0/0 -> Secondary FortiGate
! Gi0/1 -> Primary FortiGate
! Gi0/2 -> HQ-ACC-USER switch
! Gi0/3 -> HQ-ACC-SRV switch
interface range GigabitEthernet0/0 - 3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30
 no shutdown
exit

end
write memory
